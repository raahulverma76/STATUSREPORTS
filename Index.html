// ====== Google Apps Script API (Sheet se linked) ====== const SHEET_ID = '1hEXq3f1MZt5UMgWl_6bIlQZvKuGdg7QJjcPNDxTPac4'; const SHEET_NAME = 'employee'; function getSS_() { return SpreadsheetApp.openById(SHEET_ID); } function getSheet_() { return getSS_().getSheetByName(SHEET_NAME); } function json_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); } function toIso_(v, tz) { if (v instanceof Date) return Utilities.formatDate(v, tz, 'yyyy-MM-dd'); const s = String(v).trim(); let m = s.match(/^(\d{1,2})[-\/.](\d{1,2})[-\/.](\d{4})$/); if (m) return m[3] + '-' + ('0' + m[2]).slice(-2) + '-' + ('0' + m[1]).slice(-2); m = s.match(/^(\d{4})-(\d{2})-(\d{2})/); return m ? m[0] : ''; } function doGet(e) { try { const sh = getSheet_(); const tz = getSS_().getSpreadsheetTimeZone(); const last = sh.getLastRow(); if (last < 2) return json_([]); const rows = sh.getRange(2, 1, last - 1, 6).getValues(); const out = rows.filter(r => r[0] !== '' && r[1] !== '').map(r => ({ player: String(r[0]).trim().toUpperCase(), date: toIso_(r[1], tz), month: String(r[2]).trim().toUpperCase(), resume: Number(r[3]) || 0, interviews: Number(r[4]) || 0, joining: Number(r[5]) || 0 })).filter(r => r.date); return json_(out); } catch (err) { return json_({ error: String(err) }); } } function doPost(e) { const lock = LockService.getScriptLock(); try { lock.waitLock(10000); const d = JSON.parse(e.postData.contents); const sh = getSheet_(); const tz = getSS_().getSpreadsheetTimeZone(); const p = d.date.split('-'); const dt = new Date(Number(p[0]), Number(p[1]) - 1, Number(p[2])); const month = Utilities.formatDate(dt, tz, 'MMMM').toUpperCase(); sh.appendRow([String(d.player).trim().toUpperCase(), dt, month, Number(d.resume) || 0, Number(d.interviews) || 0, Number(d.joining) || 0]); sh.getRange(sh.getLastRow(), 2).setNumberFormat('dd-MM-yyyy'); return json_({ ok: true, month: month }); } catch (err) { return json_({ ok: false, error: String(err) }); } finally { lock.releaseLock(); } }
